{{- /* atom/linked.html — Emit repeated value elements for a page's terms in a given taxonomy. The value is either the term's display name (mode "term") or its absolute URL (mode "href"), depending on whether the taxonomy has a corresponding feed that consumers can join against. The element is built as a string and emitted with safeHTML: interpolating a variable into an element-NAME position otherwise trips html/template's contextual escaper (producing "ZgotmplZ"). Input (dict): "page" — the Page object "taxonomy" — taxonomy name, e.g. "software", "languages", "topics" "element" — vr: local element name to emit, e.g. "software" "mode" — "term" (emit display name) or "href" (emit absolute URL) */ -}} {{- $page := .page -}} {{- $taxonomy := .taxonomy -}} {{- $element := .element -}} {{- $mode := .mode -}} {{- range $page.GetTerms $taxonomy -}} {{- if eq $mode "term" -}} {{ printf "%s" $element .LinkTitle $element | safeHTML }} {{ else -}} {{ printf "%s" $element (.RelPermalink | strings.TrimPrefix "/" | strings.TrimSuffix "/") $element | safeHTML }} {{ end -}} {{- end -}}
