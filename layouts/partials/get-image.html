{{- /* Returns the image resource and alt text for a page. Usage: {{ $result := partial "get-image.html" $page }} {{ $result := partial "get-image.html" (dict "page" $page "fallback" true) }} {{ $result.image }} — image resource or false {{ $result.alt }} — alt text (from image-alt param or title) With fallback: true, checks parent section's image/image-alt params. Alt text is sourced from wherever the image came from. */ -}} {{- $page := . -}} {{- $fallback := false -}} {{- if reflect.IsMap . -}} {{- $page = .page -}} {{- $fallback = .fallback | default false -}} {{- end -}} {{- $image := false -}} {{- $alt := $page.Title -}} {{- $color := "" -}} {{- $software_images := slice -}} {{- /* Special handling for cheatsheets: collect all software logos and colors */ -}} {{- if eq $page.Type "resources" -}} {{- if eq $page.Params.resource_type "cheatsheet" -}} {{- with $page.Params.software -}} {{- range . -}} {{- $softwareName := . -}} {{- $softwarePath := printf "/software/%s" ($softwareName | urlize) -}} {{- $softwarePage := site.GetPage $softwarePath -}} {{- with $softwarePage -}} {{- with .Params.image -}} {{- $swImage := $softwarePage.Resources.Get . -}} {{- if $swImage -}} {{- $software_images = $software_images | append (dict "image" $swImage "color" $softwarePage.Params.color "alt" $softwarePage.Title) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Use first software for backward compatibility */ -}} {{- if ge (len $software_images) 1 -}} {{- $first := index $software_images 0 -}} {{- $image = $first.image -}} {{- $alt = $first.alt -}} {{- $color = $first.color -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 1. Check page's .Params.image (skip if we already got a cheatsheet software image) */ -}} {{- if not $image -}} {{- with $page.Params.image -}} {{- $image = $page.Resources.Get . -}} {{- if $image -}} {{- $alt = index $page.Params "image-alt" | default $page.Title -}} {{- $color = $page.Params.color -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 2. Fallback: check parent's .Params.image */ -}} {{- if and (not $image) $fallback -}} {{- $parent := $page.Parent -}} {{- with $parent -}} {{- with .Params.image -}} {{- $image = $parent.Resources.Get . -}} {{- if $image -}} {{- $alt = index $parent.Params "image-alt" | default $parent.Title -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "image" $image "alt" $alt "color" $color "software_images" $software_images) -}}
